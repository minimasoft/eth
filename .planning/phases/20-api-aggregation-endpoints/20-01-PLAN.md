```python
class DocumentTokenUsage(BaseModel):
    """Per-document token usage aggregation response."""

    has_data: bool = False
    """Whether this document has any llm_usage records."""

    prompt_tokens: int = 0
    """Total prompt (input) tokens across all LLM calls for this document."""

    completion_tokens: int = 0
    """Total completion (output) tokens across all LLM calls."""

    total_tokens: int = 0
    """Total tokens (prompt + completion) across all LLM calls."""

    cached_tokens: int = 0
    """Total cached tokens across all LLM calls (0 when not reported)."""

    total_cost: float | None = None
    """Total monetary cost across all LLM calls (None when cost data absent)."""

    duration_ms: int = 0
    """Total wall-clock duration of all LLM calls for this document."""
```

### Task 2: Add GET /documents/{id}/tokens endpoint

**File:** `src/eth_pipeline/api/routes/documents.py`

Add new route before the documents list endpoint:

```python
@router.get(
    "/documents/{document_id}/tokens",
    response_model=DocumentTokenUsage,
)
async def get_document_tokens(document_id: str) -> DocumentTokenUsage:
    """Retrieve aggregated token usage for a single document."""
    db: AsyncWsSurrealConnection | None = app.state.db
    if db is None:
        raise HTTPException(status_code=503, detail="SurrealDB unavailable")

    doc_id_obj = RecordID("document", document_id)

    try:
        result = await db.query(
            "SELECT math::sum(prompt_tokens) as prompt_tokens, "
            "math::sum(completion_tokens) as completion_tokens, "
            "math::sum(total_tokens) as total_tokens, "
            "math::sum(cached_tokens) as cached_tokens, "
            "math::sum(cost) as total_cost, "
            "math::sum(duration_ms) as duration_ms, "
            "count() as record_count "
            "FROM llm_usage WHERE document = $doc GROUP ALL",
            {"doc": doc_id_obj},
        )
        records = [r for r in (result or []) if isinstance(r, dict)]
    except Exception as exc:
        logger.error("Failed to query token usage: %s", exc)
        raise HTTPException(status_code=502, detail="Failed to query token usage")

    if not records or records[0].get("record_count", 0) == 0:
        return DocumentTokenUsage(has_data=False)

    row = records[0]
    return DocumentTokenUsage(
        has_data=True,
        prompt_tokens=row.get("prompt_tokens") or 0,
        completion_tokens=row.get("completion_tokens") or 0,
        total_tokens=row.get("total_tokens") or 0,
        cached_tokens=row.get("cached_tokens") or 0,
        total_cost=row.get("total_cost"),
        duration_ms=row.get("duration_ms") or 0,
    )
```

### Task 3: Add token aggregation to GET /documents list endpoint

**File:** `src/eth_pipeline/api/routes/documents.py`

In the existing `list_documents` function, after the main query and before building `DocumentListItem` objects, add a batched token aggregation:

```python
# After the main documents query and count query, before building items:
if document_rids:
    token_result = await db.query(
        "SELECT document, "
        "math::sum(prompt_tokens) as prompt_tokens, "
        "math::sum(completion_tokens) as completion_tokens, "
        "math::sum(total_tokens) as total_tokens, "
        "math::sum(cached_tokens) as cached_tokens, "
        "math::sum(cost) as total_cost, "
        "math::sum(duration_ms) as duration_ms "
        "FROM llm_usage "
        "WHERE document INSIDE $docs "
        "GROUP BY document",
        {"docs": document_rids},
    )
    token_rows = [r for r in (token_result or []) if isinstance(r, dict)]
    token_map: dict[str, dict] = {}
    for row in token_rows:
        doc_str = str(row.get("document", ""))
        token_map[doc_str] = row
else:
    token_map = {}
```

Then when building each `DocumentListItem`, look up token data from `token_map`:
```python
doc_key = f"document:{doc_id}"  # or str(record_id)
token_data = token_map.get(doc_key, {})
DocumentListItem(

    prompt_tokens=token_data.get("prompt_tokens") or 0,
    completion_tokens=token_data.get("completion_tokens") or 0,
    total_tokens=token_data.get("total_tokens") or 0,
    cached_tokens=token_data.get("cached_tokens") or 0,
    total_cost=token_data.get("total_cost"),
    duration_ms=token_data.get("duration_ms") or 0,

```
