    **C. LLM Calls JS functions** (insert after `fetchLogsTokens(docId)` function ending at line 2397, before the `// Logs details toggle` comment at line 2399):

    ```js
    // ===== LLM Calls functions =====

    async function fetchLlmCalls(docId, page) {
      if (llmCallsLoadingFlag) return;
      llmCallsLoadingFlag = true;
      var loader = deferredLoading('llmCalls', function() {
        document.getElementById('llm-calls-loading').style.display = 'flex';
        document.getElementById('llm-calls-table-container').style.display = 'none';
        document.getElementById('llm-calls-empty').style.display = 'none';
        document.getElementById('llm-calls-pagination').style.display = 'none';
      }, function() {
        document.getElementById('llm-calls-loading').style.display = 'none';
      });

      var cacheKey = docId + '::' + page;
      if (llmCallsDataCache[cacheKey]) {
        renderLlmCalls(llmCallsDataCache[cacheKey]);
        llmCallsLoadingFlag = false;
        if (loader && loader.done) loader.done();
        return;
      }

      var params = new URLSearchParams();
      params.set('page', String(page));
      params.set('per_page', '20');

      try {
        var res = await fetch('/documents/' + encodeURIComponent(docId) + '/llm-calls?' + params.toString());
        if (!res.ok) {
          var detail = 'Error al obtener llamadas LLM';
          try { var errData = await res.json(); if (errData && errData.detail) detail = errData.detail; } catch (_pe) {}
          throw new Error(detail);
        }
        var data = await res.json();
        llmCallsDataCache[cacheKey] = data;
        renderLlmCalls(data);
      } catch (err) {
        console.error('fetchLlmCalls error:', err);
        if (loader && loader.done) loader.done();
        document.getElementById('llm-calls-table-container').style.display = 'none';
        document.getElementById('llm-calls-pagination').style.display = 'none';
        var emptyEl = document.getElementById('llm-calls-empty');
        emptyEl.style.display = 'block';
        emptyEl.innerHTML = '<h2>Error al cargar llamadas LLM</h2><p>' + escapeHtml(err.message) + '</p>';
      } finally {
        llmCallsLoadingFlag = false;
      }
    }

    function renderLlmCalls(data) {
      var loader = deferredLoaders['llmCalls'];
      if (loader) { clearTimeout(loader); document.getElementById('llm-calls-loading').style.display = 'none'; }

      var tbody = document.getElementById('llm-calls-tbody');
      var summaryEl = document.getElementById('llm-calls-summary');
      var tableContainer = document.getElementById('llm-calls-table-container');
      var emptyEl = document.getElementById('llm-calls-empty');
      var paginationEl = document.getElementById('llm-calls-pagination');

      if (!data || !data.items || data.items.length === 0) {
        tableContainer.style.display = 'none';
        paginationEl.style.display = 'none';
        summaryEl.style.display = 'none';
        emptyEl.style.display = 'block';
        emptyEl.innerHTML = '<h2>Sin registro de llamadas LLM</h2><p>Documento anterior a v6.1</p>';
        return;
      }

      emptyEl.style.display = 'none';
      tableContainer.style.display = '';
      paginationEl.style.display = 'flex';

      tbody.innerHTML = '';
      data.items.forEach(function(item) {
        var tr = document.createElement('tr');
        tr.style.cursor = 'pointer';
        tr.innerHTML =
          '<td class="llm-call-col-model">' + escapeHtml(item.model || '—') + '</td>' +
          '<td><span class="entity-type-label">' + escapeHtml(item.activity_type || '—') + '</span></td>' +
          '<td class="llm-call-col-tokens">' + (item.prompt_tokens != null ? Number(item.prompt_tokens).toLocaleString() : '—') + '</td>' +
          '<td class="llm-call-col-tokens">' + (item.completion_tokens != null ? Number(item.completion_tokens).toLocaleString() : '—') + '</td>' +
          '<td class="llm-call-col-tokens">' + (item.total_tokens != null ? Number(item.total_tokens).toLocaleString() : '—') + '</td>' +
          '<td class="llm-call-col-cost">' + (item.cost != null ? '$' + Number(item.cost).toFixed(4) : '—') + '</td>' +
          '<td class="llm-call-col-duration">' + (item.duration_ms != null ? (item.duration_ms / 1000).toFixed(2) + 's' : '—') + '</td>' +
          '<td class="llm-call-col-timestamp">' + escapeHtml(formatDate(item.timestamp)) + '</td>';
        tbody.appendChild(tr);

        var detailsRow = document.createElement('tr');
        detailsRow.className = 'llm-call-details-row';
        var promptHtml = item.prompt_text ? '<pre class="llm-call-pre">' + escapeHtml(item.prompt_text) + '</pre>' : '<span style="color:#94a3b8;font-size:12px">—</span>';
        var responseHtml = item.response_text ? '<pre class="llm-call-pre">' + escapeHtml(item.response_text) + '</pre>' : '<span style="color:#94a3b8;font-size:12px">—</span>';
        detailsRow.innerHTML =
          '<td colspan="8" style="padding:0 12px 12px 12px">' +
            '<div style="margin-bottom:8px">' +
              '<div style="font-weight:600;font-size:13px;color:#475569;margin-bottom:4px">Prompt:</div>' +
              promptHtml +
            '</div>' +
            '<div>' +
              '<div style="font-weight:600;font-size:13px;color:#475569;margin-bottom:4px">Respuesta:</div>' +
              responseHtml +
            '</div>' +
          '</td>';
        tbody.appendChild(detailsRow);
      });

      var page = data.page || 1;
      var pages = data.pages || 0;

      document.getElementById('llm-calls-page-info').textContent = 'Página ' + page + ' de ' + pages;
      document.getElementById('llm-calls-prev-btn').disabled = page <= 1;
      document.getElementById('llm-calls-next-btn').disabled = page >= pages;
    }

    function toggleLlmCallRow(row) {
      var detailsRow = row.nextElementSibling;
      if (!detailsRow || !detailsRow.classList.contains('llm-call-details-row')) return;
      var isOpen = detailsRow.classList.contains('open');
      detailsRow.classList.toggle('open');
      row.classList.toggle('expanded', !isOpen);
    }

    async function fetchLlmCallsSummary(docId) {
      if (!docId) return;
      if (llmCallsSummaryCache[docId]) {
        updateLlmCallsSummary(llmCallsSummaryCache[docId]);
        return;
      }
      try {
        var res = await fetch('/documents/' + encodeURIComponent(docId) + '/llm-calls/summary');
        if (!res.ok) {
          document.getElementById('llm-calls-summary').style.display = 'none';
          return;
        }
        var data = await res.json();
        llmCallsSummaryCache[docId] = data;
        updateLlmCallsSummary(data);
      } catch (err) {
        console.error('fetchLlmCallsSummary error:', err);
        document.getElementById('llm-calls-summary').style.display = 'none';
      }
    }

    function updateLlmCallsSummary(data) {
      var summaryEl = document.getElementById('llm-calls-summary');
      if (!data || !data.has_data) {
        summaryEl.style.display = 'none';
        return;
      }
      document.getElementById('llm-sum-calls').textContent = (data.total_calls || 0).toLocaleString();
    ```
