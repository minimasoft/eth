    ```javascript
    const tabs = document.querySelectorAll('nav button');
    const sections = {
      'upload': document.getElementById('tab-upload'),
      'documents': document.getElementById('tab-documents'),
      'entities': document.getElementById('tab-entities'),
    };

    function switchTab(tabName) {
      // Hide all sections, deactivate all tabs
      Object.values(sections).forEach(s => s.style.display = 'none');
      tabs.forEach(t => t.classList.remove('active'));
      // Show target section, activate target tab
      sections[tabName].style.display = 'block';
      document.querySelector(`nav button[data-tab="${tabName}"]`).classList.add('active');
    }

    // Attach click handlers
    tabs.forEach(tab => {
      tab.addEventListener('click', () => switchTab(tab.dataset.tab));
    });

    // Activate Upload on load
    switchTab('upload');
    ```
