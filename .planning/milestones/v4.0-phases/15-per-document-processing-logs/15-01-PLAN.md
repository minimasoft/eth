    **Class design:**

    ```python
    class ProcessingLogger:
        """Fire-and-forget per-document processing audit logger.

        Each log() call opens its own SurrealDB connection, writes
        one entry, and closes.  This is safe for Temporal activities
        — no shared state, no replay contamination.

        Parameters
        ----------
        db_params:
            SurrealDB connection parameters dict (url, user, password, ns, database)
            as produced by activities._db_params().
        """

        def __init__(self, db_params: dict):
            self._db_params = db_params
            self._seq_counter: dict[str, int] = {}
            # _seq_counter keyed by (document_id, step_name) tuple str
    ```

    **Method: `async def log(self, document_id, step_name, severity, message, details=None)`**

    1. **Validate severity** — must be one of `"info"`, `"warning"`, `"error"` (per schema ASSERT). If invalid, default to `"info"` and log a Python warning.

    2. **Compute sequence number** — use an internal per-instance dict `_seq_counter` keyed by `f"{document_id}:{step_name}"`. Increment before using:
       ```python
       key = f"{document_id}:{step_name}"
       seq = self._seq_counter.get(key, 0)
       self._seq_counter[key] = seq + 1
       ```

    3. **Compute deterministic record ID** (per D-09):
       ```python
       import hashlib
       raw = f"{document_id}{step_name}{seq}"
       record_id = hashlib.sha256(raw.encode("utf-8")).hexdigest()[:16]
       ```
       The first 16 hex characters of the SHA256 hash produce a 64-bit identifier — unique enough for per-document scoping.

    4. **Open SurrealDB connection** using `get_db(**self._db_params)` from `eth_pipeline.db`:
       ```python
       from eth_pipeline.db import get_db
       async with get_db(**self._db_params) as db:
       ```

    5. **Enforce 100-entry cap** (per D-08) — query count of existing entries for this document:
       ```python
       count_result = await db.query(
           "SELECT count() AS total FROM document_event_log "
           "WHERE document = $doc_ref GROUP ALL",
           {"doc_ref": f"document:{document_id}"},
       )
       # Use _parse_count() logic inline (extract from flat result list)
       ```
       If count >= 100: log a Python warning `"Processing log cap reached for document {document_id} — skipping entry"` and return without writing. Use `import logging; logger = logging.getLogger(__name__)` for this.

    6. **Write the log entry** using `UPSERT` (deterministic ID → idempotent on retry):
       ```python
       doc_record = RecordID("document", document_id)
       # Use raw SURQL CREATE with explicit record ID for deterministic insert
       await db.query(
           "CREATE document_event_log:$rid CONTENT { "
           "document: $doc, "
           "step_name: $step, "
           "severity: $sev, "
           "message: $msg, "
           "details: $det "
           "}",
           {
               "rid": record_id,
               "doc": doc_record,
               "step": step_name,
               "sev": severity,
               "msg": message,
               "det": details,
           },
       )
       ```
       Using `CREATE document_event_log:$rid` with a deterministic record ID ensures that Temporal replay (which produces the same sequence of log calls) overwrites the same records — no duplicates, no orphaned entries.

       Note: `CREATE` with an explicit record ID will fail on first retry if the record already exists. Use `UPSERT` semantics instead. The proper SurrealDB pattern is:
       ```python
       await db.query(
           "UPDATE document_event_log:$rid CONTENT { "
           "document: $doc, step_name: $step, "
           "severity: $sev, message: $msg, details: $det, "
           "created_at: time::now() "
           "}",
           ...
       )
       ```
       Use `UPDATE ... CONTENT` which acts as UPSERT — creates if absent, overwrites if present. Set `created_at: time::now()` so replay resets the timestamp to match the retry time.

    7. **Wrap the entire body (steps 4-6) in try/except** — fire-and-forget means the activity never waits for the log write to succeed. If SurrealDB is unreachable, catch any exception and log a warning via Python's `logging.getLogger(__name__)`.

    **Error handling pattern for SurrealDB failures (fire-and-forget):**
    ```python
    try:
        async with get_db(...) as db:
            # count check + write
    except ConnectionError:
        logger.warning("ProcessingLogger: SurrealDB unavailable for document %s", document_id)
    except Exception as exc:
        logger.warning("ProcessingLogger: write failed for document %s: %s", document_id, exc)
    ```

    **Module-level exports:**
    ```python
    __all__ = ["ProcessingLogger"]
    ```

    **No imports from activities.py or any Temporal module.** This module is purely
    about DB writes. Standard library + `eth_pipeline.db` + `surrealdb.data.types.record_id`.

    ```python
    class ProcessingLogListItem(BaseModel):
        """A single log entry in the paginated log list."""

        id: str
        """Deterministic log entry ID (SHA256 hex digest, first 16 chars)."""

        document_id: str
        """ID of the document this log entry belongs to."""

        step_name: str
        """Processing step that created this entry."""

        severity: str
        """Severity level: info, warning, or error."""

        message: str
        """Human-readable log message."""

        details: dict | None = None
        """Optional structured metadata attached to this entry."""

        created_at: str | None = None
        """ISO-8601 timestamp of when the log entry was created."""


    class ProcessingLogListResponse(BaseModel):
        """Paginated response body for GET /documents/{id}/logs."""

        items: list[ProcessingLogListItem]
        """List of log entries on the current page."""

        total: int
        """Total number of log entries for this document."""

        page: int
        """Current page number (1-based)."""

        per_page: int
        """Number of items per page (fixed at 50)."""

        pages: int
        """Total number of pages available."""
    ```

    **2. Add endpoint** (after the `clear_document_events` endpoint, before the
    `delete_document` endpoint):

    ```python
    # =======================================================================
    # Get document processing logs (paginated)
    # =======================================================================
    @app.get(
        "/documents/{document_id}/logs",
        response_model=ProcessingLogListResponse,
    )
    async def get_document_logs(
        document_id: str,
        page: int = Query(1, ge=1),
    ) -> ProcessingLogListResponse:
        """Retrieve processing log entries for a document (paginated, newest first).

        Returns log entries ordered by created_at DESC (newest first),
        50 entries per page (fixed per D-04). Uses the same pagination
        envelope as /documents and /entities endpoints.

        Returns HTTP 404 if the document does not exist and HTTP 503 if the
        database is unavailable.
        """
        db: AsyncWsSurrealConnection | None = app.state.db

        if db is None:
            logger.error(
                "GET /documents/%s/logs rejected — SurrealDB unavailable",
                document_id,
            )
            raise HTTPException(
                status_code=503,
                detail="SurrealDB is not available. Please try again later.",
            )

        per_page = 50  # Fixed per D-04
        offset = (page - 1) * per_page
        doc_ref = f"document:{document_id}"

        # Verify document exists — use RecordID parameter
        try:
            from surrealdb.data.types.record_id import RecordID

            doc_id_obj = RecordID("document", document_id)
            exists_result = await db.query(
                "SELECT * FROM document WHERE id = $doc_id",
                {"doc_id": doc_id_obj},
            )
        except Exception as exc:
            logger.error(
                "Failed to query document %s: %s", document_id, exc
            )
            raise HTTPException(
                status_code=502,
                detail="Failed to query database.",
            ) from exc

        exists_records: list[dict] = [
            r for r in (exists_result or []) if isinstance(r, dict)
        ]
        if not exists_records:
            logger.warning("Document %s not found for log query", document_id)
            raise HTTPException(
                status_code=404,
                detail=f"Document {document_id} not found.",
            )

        # Count total log entries for this document
        try:
            count_result = await db.query(
                "SELECT count() AS total FROM document_event_log "
                "WHERE document = $doc_ref GROUP ALL",
                {"doc_ref": doc_ref},
            )
        except Exception as exc:
            logger.error(
                "Failed to count log entries for document %s: %s",
                document_id,
                exc,
            )
            raise HTTPException(
                status_code=502,
                detail="Failed to query database.",
            ) from exc

        total = _parse_count(count_result)

        if total == 0:
            pages = 0
        else:
            pages = max(1, (total + per_page - 1) // per_page)

        # Fetch paginated log entries (newest first per D-04)
        try:
            data_result = await db.query(
                "SELECT * FROM document_event_log "
                "WHERE document = $doc_ref "
                "ORDER BY created_at DESC "
                "LIMIT $limit START $offset",
                {
                    "doc_ref": doc_ref,
                    "limit": per_page,
                    "offset": offset,
                },
            )
        except Exception as exc:
            logger.error(
                "Failed to query log entries for document %s: %s",
                document_id,
                exc,
            )
            raise HTTPException(
                status_code=502,
                detail="Failed to query database.",
            ) from exc

        from surrealdb.data.types.record_id import RecordID

        data_records: list[dict] = [
            r for r in (data_result or []) if isinstance(r, dict)
        ]

        items: list[ProcessingLogListItem] = []
        for record in data_records:
            # Parse record_id from SurrealDB RecordID
            entry_id_val = record.get("id")
            entry_id: str = ""
            if isinstance(entry_id_val, RecordID):
                entry_id = str(entry_id_val.id)
            elif isinstance(entry_id_val, str):
                entry_id = (
                    entry_id_val.split(":", 1)[1]
                    if ":" in entry_id_val

                )

                details=record.get("details"),
                created_at=created_at_str,

            "Listed processing logs for document %s (page=%d) — %d items of %d total",
            document_id,
            page,
            len(items),
            total,
        )

        return ProcessingLogListResponse(
            items=items,
            total=total,
            page=page,
            per_page=per_page,
            pages=pages,
        )
    ```
